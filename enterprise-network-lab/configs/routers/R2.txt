! ============================================================
! R2 — Building 2 WAN Router
! Model   : Cisco 2901 (with HWIC-2T for serial)
! Role    : WAN gateway, OSPF speaker
! Links   : Gi0/1 → MSW3, Gi0/0 → MSW4, Se0/0/0 → R1 (DTE)
! ============================================================

hostname R2

interface GigabitEthernet0/1
 description Routed-to-MSW3-Gi0/1
 ip address 10.10.2.1 255.255.255.252
 no shutdown

interface GigabitEthernet0/0
 description Routed-to-MSW4-Gi0/1
 ip address 10.10.2.5 255.255.255.252
 no shutdown

interface Serial0/0/0
 description WAN-to-R1 (DTE end)
 ip address 10.10.10.2 255.255.255.252
 no shutdown

interface Serial0/0/1
 no ip address
 shutdown

interface Vlan1
 no ip address
 shutdown

! --- OSPF ---
router ospf 1
 log-adjacency-changes
 network 10.10.2.0 0.0.0.3 area 0
 network 10.10.2.4 0.0.0.3 area 0
 network 10.10.10.0 0.0.0.3 area 0

! --- NTP ---
ntp server 10.10.1.1

line con 0
line vty 0 4
 login
